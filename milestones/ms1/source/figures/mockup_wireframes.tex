\begin{tikzpicture}[
    x=1cm, y=1cm,
    frame/.style={draw=gray!70, rounded corners=3pt, fill=white},

\colorlet{colA}{RoyalBlue}
\colorlet{colB}{Orange}

% ---------- reusable window chrome: #1 = x, #2 = y, #3 = label ----------
\def\screen#1#2#3{%
  \draw[frame] (#1,#2) rectangle ++(8,4.4);
  \draw[gray!70] (#1,#2+4.05) -- ++(8,0);
  \foreach \k in {0,1,2} \fill[gray!45] (#1+0.2+0.18*\k,#2+4.225) circle (0.05);
  \node[font=\scriptsize\bfseries, gray!80!black] at (#1+4,#2+4.225) {#3};
}
% ---------- fit bar: #1 = x, #2 = y, #3 = filled, #4 = colour ----------
\def\fitbar#1#2#3#4{%
  \foreach \k in {1,...,5}{
    \ifnum\k>#3 \draw[#4] (#1+0.1+0.2*\k,#2-0.07) rectangle ++(0.14,0.14);
    \else \fill[#4] (#1+0.1+0.2*\k,#2-0.07) rectangle ++(0.14,0.14); \fi }
}
